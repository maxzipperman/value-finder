    result = {}  # book -> row

                    if idx < prev_idx:

                        best_index[book] = idx

    return result


def _parse_iso_datetime(s):
    """
    Parse an ISO 8601 datetime string with timezone info.
    Supports 'Z' suffix and +/-HH:MM offsets.
    """
    if not isinstance(s, str) or not s:
        raise ValueError("Invalid string")

    # Check for Z suffix
    if s.endswith('Z'):
        base = s[:-1]
        tz_offset = datetime.timezone.utc
    else:
        # Find the timezone part. It should be at the end and start with + or -
        # Format: YYYY-MM-DDTHH:MM:SS[.ffffff][+HH:MM|-HH:MM]
        # We need to find the last occurrence of + or - that is not part of the date/time
        # Standard ISO format has T between date and time.
        
        # Split by T to separate date and time parts roughly, but timezone can be attached to time
        # Better approach: look for +/- at the end after the seconds/minutes
        
        # Find the last + or - that is followed by digits and possibly :
        # Regex-like manual parsing
        
        tz_part = None
        base = s
        
        # Check if there is a timezone offset at the end
        # It must be +HH:MM, +HHMM, +HH, -HH:MM, -HHMM, -HH
        # Or just Z which we handled.
        
        # Find the last occurrence of '+' or '-' that could be a timezone marker
        # It should be after the 'T' if present, or after the date if no time? 
        # ISO 8601 usually has T. Let's assume standard format.
        
        # Locate the timezone offset
        idx_plus = s.rfind('+')
        idx_minus = s.rfind('-')
        
        # We want the last one that is likely a timezone offset.
        # If both exist, the last one is usually the timezone if it's at the end.
        # But '-' can appear in dates? No, dates use '-'. Time uses ':'.
        # The timezone offset starts with + or - and ends the string.
        
        tz_idx = -1
        if idx_plus >= 0 and idx_minus >= 0:
            tz_idx = max(idx_plus, idx_minus)
        elif idx_plus >= 0:
            tz_idx = idx_plus
        elif idx_minus >= 0:
            tz_idx = idx_minus
        
        if tz_idx == -1 or tz_idx < len(s) - 5: # Minimum offset length is +HH:MM (6 chars) or +HHMM (5 chars)
             # If the offset is too short or not at the end, it might be invalid or part of the time?
             # Actually, if there's no T, it might be just date. But we need datetime.
             # Let's assume standard ISO 8601 datetime string.
             pass
        
        # If we found a potential tz marker at the end
        if tz_idx >= 0 and (s[tz_idx:].replace(':', '').isdigit() or s[tz_idx:].replace(':', '').lstrip('+').isdigit()):
            # Validate it looks like an offset
            offset_str = s[tz_idx:]
            base = s[:tz_idx]
            
            sign = 1 if offset_str[0] == '+' else -1
            rest = offset_str[1:]
            
            if ':' in rest:
                parts = rest.split(':')
                if len(parts) != 2:
                    raise ValueError("Invalid timezone format")
                hh, mm = int(parts[0]), int(parts[1])
            else:
                # Could be HHMM or HH
                if len(rest) == 4:
                    hh, mm = int(rest[:2]), int(rest[2:])
                elif len(rest) == 2:
                    hh, mm = int(rest), 0
                else:
                    raise ValueError("Invalid timezone format")
            
            tz_offset = datetime.timezone(datetime.timedelta(hours=sign * hh, minutes=sign * mm))
        else:
            # No valid timezone offset found at the end
            raise ValueError("Missing timezone info")

    # Parse the base datetime string
    # Format: YYYY-MM-DDTHH:MM:SS[.ffffff]
    try:
        # Try parsing with microseconds
        if '.' in base:
            dt = datetime.datetime.strptime(base, "%Y-%m-%dT%H:%M:%S.%f")
        else:
            dt = datetime.datetime.strptime(base, "%Y-%m-%dT%H:%M:%S")
    except ValueError:
        raise ValueError("Invalid datetime format")

    # Attach timezone
    dt = dt.replace(tzinfo=tz_offset)
    
    return dt
