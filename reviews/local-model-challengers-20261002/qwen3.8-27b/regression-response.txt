def run_tests(candidate):
    def check_valid(value, expected):
        try:
            result = candidate(value)
        except Exception as e:
            raise AssertionError(
                f"candidate({value!r}) raised {type(e).__name__}: {e}"
            )
        if result != expected:
            raise AssertionError(
                f"candidate({value!r}) = {result!r}, expected {expected!r}"
            )
        if not isinstance(result, float):
            raise AssertionError(
                f"candidate({value!r}) returned {type(result).__name__}, expected float"
            )

        try:
            candidate(value)

            )
        else:
            raise AssertionError(

            )
