from datetime import datetime, timezone

def run_tests(candidate):
    """
    Regression tests for the parse_utc function.
    Validates:
    - None/blank/invalid input handling.
    - Whitespace stripping.
    - ISO Z format.
    - Explicit offsets.
    - Compact UTC (HHMMZ).
    - Naive ISO interpreted as UTC.
    - Aware UTC normalization.
    - Preservation of fractional seconds.
    """
    UTC = timezone.utc

    # 1. Null/Blank/Invalid Inputs
    assert candidate(None) is None
    assert candidate("") is None
    assert candidate("   ") is None
    assert candidate("not-a-date") is None
    assert candidate("2026-13-01") is None  # Invalid month

    # 2. Whitespace Stripping
    assert candidate("  2026-09-29T14:30:07Z  ") == datetime(2026, 9, 29, 14, 30, 7, tzinfo=UTC)

    # 3. Standard ISO Z
    assert candidate("2026-09-29T14:30:07Z") == datetime(2026, 9, 29, 14, 30, 7, tzinfo=UTC)

    # 4. Explicit Offsets (Normalization to UTC)
    # +02:00 -> 12:00 UTC
    assert candidate("2026-10-02T14:00:00+02:00") == datetime(2026, 10, 2, 12, 0, 0, tzinfo=UTC)
    # -05:00 -> 19:00 UTC
    assert candidate("2026-10-02T14:00:00-05:00") == datetime(2026, 10, 2, 19, 0, 0, tzinfo=UTC)

    # 5. Compact UTC (HHMMZ)
    # Testing formats like 2026-09-29T0002Z
    assert candidate("2026-09-29T0002Z") == datetime(2026, 9, 29, 0, 2, tzinfo=UTC)
    # Testing formats like 2026-09-29T00:02Z (if supported by implementation)
    assert candidate("2026-09-29T00:02Z") == datetime(2026, 9, 29, 0, 2, tzinfo=UTC)

    # 6. Naive ISO interpreted as UTC
    # "2026-09-29T14:30:07" (no Z, no offset)
    assert candidate("2026-09-29T14:30:07") == datetime(2026, 9, 29, 14, 30, 7, tzinfo=UTC)
    assert candidate("2026-09-29 14:30:07") == datetime(2026, 9, 29, 14, 30, 7, tzinfo=UTC)

    # 7. Fractional Seconds Preservation
    # ISO 8601 allows fractional seconds
    res_frac = candidate("2026-09-29T14:30:07.123456Z")
    assert res_frac == datetime(2026, 9, 29, 14, 30, 7, 123456, tzinfo=UTC)
    
    res_frac_short = candidate("2026-09-29T14:30:07.5Z")
    assert res_frac_short == datetime(2026, 9, 29, 14, 30, 7, 500000, tzinfo=UTC)

    # 8. Edge Case: Leap Year / End of Year
    assert candidate("2024-02-29T12:00:00Z") == datetime(2024, 2, 29, 12, 0, 0, tzinfo=UTC)
    assert candidate("2023-12-31T23:59:59Z") == datetime(2023, 12, 31, 23, 59, 59, tzinfo=UTC)
