    UTC = timezone.utc

    # Helper to assert equality of datetimes
    def assert_eq(actual, expected, msg=""):
        if actual is None and expected is None:
            return
        if actual is None or expected is None:
            raise AssertionError(f"{msg}: expected {expected}, got {actual}")
        if actual != expected:
            raise AssertionError(f"{msg}: expected {expected}, got {actual}")
        # Also check tzinfo is UTC
        if actual.tzinfo is not UTC:
            raise AssertionError(f"{msg}: expected tzinfo UTC, got {actual.tzinfo}")

    assert_eq(candidate(None), None, "None input")

    assert_eq(candidate(""), None, "Empty string")

    assert_eq(candidate("   "), None, "Whitespace only")

    expected = datetime(2026, 9, 29, 14, 30, 7, tzinfo=UTC)
    assert_eq(candidate("2026-09-29T14:30:07Z"), expected, "Standard ISO Z")

    assert_eq(candidate("  2026-09-29T14:30:07Z  "), expected, "ISO Z with whitespace")
