```python
import math


def run_tests(candidate):
    """
    Test suite for american_probability.
    
    american_probability(value) returns implied probability:
      - positive odds => 100/(value+100)
      - negative odds => -value/(-value+100)
    
    Accepts: int, float, numeric strings (with surrounding whitespace)
    Only finite values with absolute value >= 100.
    Rejects: bool, None, nonnumeric strings, all other types with ValueError.
    Does not round output.
    """
    
    def _check_value(candidate, input_val, expected, should_raise=False, exc_type=ValueError):
        """Helper to check a single value."""
        if should_raise:
            try:
                result = candidate(input_val)
                raise AssertionError(f"Expected {exc_type.__name__} for input {input_val!r}, but got result {result!r}")
            except exc_type:
                pass
            except Exception as e:
                raise AssertionError(f"Expected {exc_type.__name__} for input {input_val!r}, but got {type(e).__name__}: {e}")
        else:
            try:
                result = candidate(input_val)
            except Exception as e:
                raise AssertionError(f"Unexpected exception {type(e).__name__}: {e} for input {input_val!r}")
            # Check that result is a float (or numeric) and matches expected
            if not isinstance(result, (int, float)):
                raise AssertionError(f"Expected numeric result, got {type(result).__name__} for input {input_val!r}")
            # Use math.isclose for floating point comparison
            if not math.isclose(result, expected, rel_tol=1e-12, abs_tol=1e-15):
                raise AssertionError(f"Expected {expected!r} for input {input_val!r}, but got {result!r}")
    
    # --- Valid inputs: positive odds ---
    # 100 => 100/(100+100) = 0.5
    _check_value(candidate, 100, 0.5)
    
    # 200 => 100/(200+100) = 100/300 = 1/3
    _check_value(candidate, 200, 1.0/3.0)
    
    # 500 => 100/(500+100) = 100/600 = 1/6
    _check_value(candidate, 500, 1.0/6.0)
    
    # 1000 => 100/(1000+100) = 100/1100 = 1/11
    _check_value(candidate, 1000, 1.0/11.0)
    
    # --- Valid inputs: negative odds ---
    # -100 => -(-100)/(-(-100)+100) = 100/(100+100) = 0.5
    _check_value(candidate, -100, 0.5)
    
    # -200 => -(-200)/(-(-200)+100) = 200/(200+100) = 200/300 = 2/3
    _check_value(candidate, -200, 2.0/3.0)
    
    # -500 => 500/(500+100) = 500/600 = 5/6
    _check_value(candidate, -500, 5.0/6.0)
    
    # -1000 => 1000/(1000+100) = 1000/1100 = 10/11
    _check_value(candidate, -1000, 10.0/11.0)
    
    # --- Valid inputs: float values ---
    # 150.0 => 100/(150+100) = 100/250 = 0.4
    _check_value(candidate, 150.0, 0.4)
    
    # -150.0 => 150/(150+100) = 150/250 = 0.6
    _check_value(candidate, -150.0, 0.6)
    
    # 123.45 => 100/(123.45+100) = 100/223.45
    _check_value(candidate, 123.45, 100.0/223.45)
    
    # -123.45 => 123.45/(123.45+100) = 123.45/223.45
    _check_value(candidate, -123.45, 123.45/223.45)
    
    # --- Valid inputs: numeric strings ---
    # "100" => 0.5
    _check_value(candidate, "100", 0.5)
    
    # " 100 " => 0.5 (whitespace allowed)
    _check_value(candidate, " 100 ", 0.5)
    
    # "  -200  " => 2/3
    _check_value(candidate, "  -200  ", 2.0/3.0)
    
    # "150.5" => 100/(150.5+100) = 100/250.5
    _check_value(candidate, "150.5", 100.0/250.5)
    
    # "-150.5" => 150.5/(150.5+100) = 150.5/250.5
    _check_value(candidate, "-150.5", 150.5/250.5)
    
    # "  500  " => 1/6
    _check_value(candidate, "  500  ", 1.0/6.0)
    
    # --- Valid inputs: boundary values (absolute value exactly 100) ---
    # 100 is already tested
    # -100 is already tested
    
    # --- Invalid inputs: should raise ValueError ---
    
    # bool values (True and False are instances of int in Python, but should be rejected)
    _check_value(candidate, True, None, should_raise=True)
    _check_value(candidate, False, None, should_raise=True)
    
    # None
    _check_value(candidate, None, None, should_raise=True)
    
    # Non-numeric strings
    _check_value(candidate, "abc", None, should_raise=True)
    _check_value(candidate, "12.34.56", None, should_raise=True)
    _check_value(candidate, "", None, should_raise=True)
    _check_value(candidate, "   ", None, should_raise=True)
    _check_value(candidate, "100abc", None, should_raise=True)
    _check_value(candidate, "abc100", None, should_raise=True)
    _check_value(candidate, "1e100", None, should_raise=True)  # Scientific notation - not a simple numeric string per typical interpretation
    _check_value(candidate, "+100", None, should_raise=True)  # Plus sign - may or may not be accepted; let's be strict
    _check_value(candidate, "100.0.0", None, should_raise=True)
    
    # Values with absolute value < 100
    _check_value(candidate, 50, None, should_raise=True)
    _check_value(candidate, -50, None, should_raise=True)
    _check_value(candidate, 99, None, should_raise=True)
    _check_value(candidate, -99, None, should_raise=True)
    _check_value(candidate, 0, None, should_raise=True)
    _check_value(candidate, -1, None, should_raise=True)
    _check_value(candidate, 1, None, should_raise=True)
    _check_value(candidate, 99.99, None, should_raise=True)
    _check_value(candidate, -99.99, None, should_raise=True)
    _check_value(candidate, "50", None, should_raise=True)
    _check_value(candidate, "-50", None, should_raise=True)
    _check_value(candidate, " 99 ", None, should_raise=True)
    _check_value(candidate, " -99 ", None, should_raise=True)
    
    # Non-finite values
    _check_value(candidate, float('inf'), None, should_raise=True)
    _check_value(candidate, float('-inf'), None, should_raise=True)
    _check_value(candidate, float('nan'), None, should_raise=True)
    _check_value(candidate, "inf", None, should_raise=True)
    _check_value(candidate, "-inf", None, should_raise=True)
    _check_value(candidate, "nan", None, should_raise=True)
    
    # Other types
    _check_value(candidate, [100], None, should_raise=True)
    _check_value(candidate, {100: 1}, None, should_raise=True)
    _check_value(candidate, (100,), None, should_raise=True)
    _check_value(candidate, b"100", None, should_raise=True)
    _check_value(candidate, complex(100, 0), None, should_raise=True)
    
    # --- Additional edge cases for valid inputs ---
    
    # Large positive value
    _check_value(candidate, 1000000, 100.0/1000100.0)
    
    # Large negative value
    _check_value(candidate, -1000000, 1000000.0/1000100.0)
    
    # String with large value
    _check_value(candidate, "1000000", 100.0/1000100.0)
    _check_value(candidate, "-1000000", 1000000.0/1000100.0)
    
    # --- Verify that the function does not round ---
    # 100/300 = 0.3333333333333333... should not be rounded to 0.33 or 0.333
    result = candidate(200)
    if result == 0.33 or result == 0.333 or result == 0.3333:
        raise AssertionError(f"Result appears to be rounded: {result!r}")
    
    # Check that 1/3 is represented with full precision
    expected_1_3 = 1.0/3.0
    if candidate(200) != expected_1_3:
        # Allow for tiny floating point differences but not rounding
        if not math.isclose(candidate(200), expected_1_3, rel_tol=1e-15, abs_tol=1e-18):
            raise AssertionError(f"Result for 200 does not match 1/3 with sufficient precision: {candidate(200)!r} vs {expected_1_3!r}")
    
    # --- Test that int and float inputs give same results ---
    _check_value(candidate, 100, 0.5)
    _check_value(candidate, 100.0, 0.5)
    _check_value(candidate, -100, 0.5)
    _check_value(candidate, -100.0, 0.5)
    
    # --- Test that string and numeric inputs give same results ---
    _check_value(candidate, 200, 1.0/3.0)
    _check_value(candidate, "200", 1.0/3.0)
    _check_value(candidate, -200, 2.0/3.0)
    _check_value(candidate, "-200", 2.0/3.0)
    
    # All tests passed
    return True
```
