import math
from datetime import datetime, timezone


def _parse_iso_utc(value):
    if not isinstance(value, str) or not value:
        return None
    text = value.strip()
    if not text:
        return None
    if text.endswith('Z') or text.endswith('z'):
        text = text[:-1] + '+00:00'
    try:
        dt = datetime.fromisoformat(text)
    except ValueError:
        return None
    if dt.tzinfo is None:
        return None
    return dt.astimezone(timezone.utc)


def _parse_odds(value):
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return float(value) if abs(value) >= 100 else None
    if isinstance(value, float):
        if not math.isfinite(value) or abs(value) < 100:
            return None
        return value
    if isinstance(value, str):
        text = value.strip()
        if not text:
            return None
        try:
            num = float(text)
        except ValueError:
            return None
        if not math.isfinite(num) or abs(num) < 100:
            return None
        return num
    return None


def select_books(rows, asof):
    if not isinstance(asof, datetime) or asof.tzinfo is None or asof.utcoffset() is None:
        raise ValueError("asof must be a timezone-aware datetime")
    asof_utc = asof.astimezone(timezone.utc)

    results = {}
    for row in rows:
        if not isinstance(row, dict):
            continue
        book = row.get('book')
        if not isinstance(book, str) or not book:
            continue
        rid = row.get('id')
        if not isinstance(rid, str) or not rid:
            continue
        odds = _parse_odds(row.get('odds'))
        if odds is None:
            continue
        observed_at = _parse_iso_utc(row.get('observed_at'))
        if observed_at is None:
            continue
        expires_at = _parse_iso_utc(row.get('expires_at'))
        if expires_at is None:
            continue
        if not (observed_at < expires_at):
            continue
        if not (observed_at <= asof_utc < expires_at):
            continue
        current = results.get(book)
        if current is None:
            results[book] = (observed_at, rid, row)
            continue
        cur_obs, cur_id, _ = current
        if (observed_at, rid) < (cur_obs, cur_id):
            results[book] = (observed_at, rid, row)

    return {book: entry[2] for book, entry in results.items()}
