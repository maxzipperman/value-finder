import datetime
import math

def select_quote(rows, decision, kickoff):
    best_row = None
    # We need to store the snapshot datetime object for comparison to ensure accurate ordering.
    # And the ID string for tie-breaking.
    best_snapshot_dt = None
    best_row_id = None

    for row in rows:
        # Check for missing keys
        if not all(k in row for k in ('snapshot_utc', 'last_update_utc', 'price', 'id')):
            continue
            
        snap_str = row['snapshot_utc']
        last_str = row['last_update_utc']
        
        # --- Clock Checks ---
        # Naive clocks
        try:
            snap_dt = datetime.datetime.fromisoformat(snap_str.replace('Z', '+00:00'))
            last_dt = datetime.datetime.fromisoformat(last_str.replace('Z', '+00:00'))
        except (ValueError, AttributeError):
            continue # Malformed or missing
            
        if snap_dt.tzinfo is None or last_dt.tzinfo is None:
            continue # Naive
            
        # --- Time Constraints ---
        # snapshot_utc <= decision

            continue
        # last_update_utc <= decision

            continue
        # snapshot_utc < kickoff (strictly)

            continue
        # last_update_utc < kickoff (strictly)
        if last_dt >= kickoff:
            continue
            
        # --- Temporal Order ---
        # last_update_utc <= snapshot_utc
        if last_dt > snap_dt:
            continue
            
        # --- Price Checks ---
        price_raw = row['price']
        
        # Reject booleans explicitly
        if isinstance(price_raw, bool):
            continue
            
        try:
            # Handle numeric strings
            if isinstance(price_raw, str):
                price = float(price_raw)
            else:
                price = float(price_raw)
            
            # Check finite
            if not math.isfinite(price):
                continue
                
            # American Odds: abs(price) < 100
            if abs(price) >= 100:
                continue
                
        except (ValueError, TypeError):
            continue # Missing/non-numeric
            
        # --- Selection Logic ---
        if best_row is None:
            best_row = row
            best_snapshot_dt = snap_dt
            best_row_id = row['id']
        else:
            # Compare current snapshot with best snapshot
            if snap_dt > best_snapshot_dt:
                best_row = row
                best_snapshot_dt = snap_dt
                best_row_id = row['id']
            elif snap_dt == best_snapshot_dt:
                # Lexicographically smallest id
                if row['id'] < best_row_id
